# 技能详细契约

本文件由 `agents/interface.yaml` 拆分而来：接口文件只保留机器可读最小集（界面元信息与兼容性声明），详细输入/输出/排除/安全/引用契约在此按需读取。

input:
  target:
    description: "目标文件路径或目录路径（.cs 文件或包含 .cs 文件的目录）"
    type: "string"
    required: true
    validation: "must exist on filesystem"
  exclude_generated:
    description: "是否排除 Generated/ 或 Gen/ 目录中的文件"
    type: "boolean"
    required: false
    default: false
  no_save:
    description: "是否跳过结果文件保存"
    type: "boolean"
    required: false
    default: false
output:
  format:
    - type: "human_readable"
      description: "按文件分组、显示成员名称和代码摘要的控制台文本"
    - type: "json"
      description: "完整结构化数据，含 File/Line/Severity/Category/RuleName/Member/MemberType/Message/CodeSnippet"
    - type: "markdown_report"
      description: ".md 文件，包含问题表格和类别分布统计"
    - type: "json_data"
      description: ".json 文件，完整结构化数据备份"
  default_save_path: "check_logs/"
  naming: "{时间戳}_{目标名称}_{问题统计}_{8位GUID}.{ext}"
boundary:
  inclusions:
    - ".cs 文件"
    - "XML 文档注释（///）检查"
    - "行内注释（//）检查"
    - "SemanticModel 语义一致性检查"
  exclusions:
    - "Generated/ 或 Gen/ 目录文件"
    - "文件头含 /// <auto-generated /> 标记的文件"
    - "排除路径配置中指定的目录"
    - "不修改源代码（只检查不修复）"
  constraints:
    - "仅检查 .cs 文件，不支持其他语言"
    - "语义分析依赖 Unity Library/ScriptAssemblies 目录"
    - "行内注释检查仅覆盖方法体"
  limitations:
    - "启发式规则（inline_magic_number、inline_condition 等）可能产生误报"
    - "inline_algorithm 和 inline_order 默认禁用"
    - "构造函数、属性访问器、运算符重载的方法体行内注释未被检查"
config:
  path: "工具目录 comment_config.json（引擎侧规则）+ 项目侧 comment_config.project.json（索引 comment.project_config），读取时深度合并"
  schema:
    exclude:
      paths: "排除的路径片段列表"
      autoGenerated: "是否跳过 auto-generated 标记文件"
    rules:
      "<rule_name>":
        enabled: "规则是否启用"
        description: "规则描述"
        issues:
          "<category>":
            enabled: "Issue 类别是否启用"
            severity: "error/warning/info"
            description: "Issue 描述"
