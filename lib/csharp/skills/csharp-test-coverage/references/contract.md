# 技能详细契约

本文件由 `agents/interface.yaml` 拆分而来：接口文件只保留机器可读最小集（界面元信息与兼容性声明），详细输入/输出/排除/安全/引用契约在此按需读取。

input:
  target:
    description: "目标包/目录/文件路径（.cs 文件或含 .cs 的目录）"
    type: "string"
    required: false
    default: "按 --scope 取默认范围"
    validation: "must exist on filesystem when provided"
  scope:
    description: "未指定 target 时的默认扫描范围"
    type: "enum"
    values: ["all", "packages", "assets"]
    required: false
    default: "all"
  real_coverage:
    description: "是否合并 Unity 真实覆盖率数据（需先执行 collect_real_coverage.py）"
    type: "boolean"
    required: false
    default: false
  no_save:
    description: "是否跳过结果文件保存"
    type: "boolean"
    required: false
    default: false
  generate_skeleton:
    description: "是否生成测试骨架（必须先经用户确认，默认 false）"
    type: "boolean"
    required: false
    default: false
output:
  format:
    - type: "human_readable"
      description: "按模块的覆盖率分布与未覆盖成员摘要"
    - type: "json"
      description: "完整结构化记录，含 CoverageRecord[]（文件/类型/成员/覆盖率状态/引用位置/建议落地方式）"
    - type: "markdown_report"
      description: ".md 报告，含概览、模块分布、未覆盖成员清单、零测试文件、弱引用、与真实覆盖率的分歧、骨架清单"
    - type: "json_data"
      description: ".json 数据文件，完整结构化记录备份"
  default_save_path: "check_logs/"
  naming: "{时间戳}_{目标名称}_{未覆盖数}u{弱引用数}r_{8位GUID}.{ext}"
boundary:
  inclusions:
    - ".cs 文件（运行时源码与测试代码）"
    - "成员级覆盖率静态分析（public + internal）"
    - "目录镜像映射与命名空间消歧"
    - "弱引用识别（XML cref、反射字符串成员名）"
    - "Unity Code Coverage 真实采集（可选）"
    - "测试骨架生成（需用户确认）"
  exclusions:
    - "Generated/ 或 Gen/ 目录的运行时文件（不计入分母）"
    - "文件头含 /// <auto-generated /> 标记的文件"
    - "Assets/Plugins/ 与 Library/PackageCache/ 三方代码"
    - "不修改运行时源码"
    - "不修复测试断言（骨架为待补完脚手架）"
  constraints:
    - "静态分析需 .NET 10.0+ SDK；首次运行自动 dotnet publish"
    - "符号解析依赖 Unity Library/ScriptAssemblies 作为程序集引用"
    - "真实采集需 unity CLI 且全程不关闭编辑器"
    - "覆盖率开关状态必须经 unity command script eval 读取，禁止读 Settings.json"
  limitations:
    - "启发式引用识别可能误报（反射字符串、动态调用）"
    - "弱引用不计入强引用覆盖，可能出现假阴性"
    - "分支覆盖率在本项目实测恒为 0，报告标注不可用"
    - "IL2CPP 与 HybridCLR 热更 dll 不在真实覆盖范围"
config:
  path: "工具目录 coverage_config.json（引擎侧通用）+ 项目侧 coverage_config.project.json（索引 coverage.project_config），读取时深度合并"
  schema:
    exclude:
      paths: "排除的路径片段列表"
      autoGenerated: "是否跳过 auto-generated 标记文件"
      autoGeneratedScanLines: "标记扫描行数"
    runtimeRoots: "运行时源码根路径前缀"
    testRootMarkers: "测试目录标识片段"
    memberScope:
      accessibilities: "纳入分母的访问级别"
      includePrivateMembers: "是否纳入 private 成员"
      skipCompilerGenerated: "是否跳过编译器生成成员"
    referenceStrength:
      weakReferenceKinds: "弱引用类别（XmlCref / ReflectionStringMemberName）"
      countWeakAsCovered: "弱引用是否计为已覆盖"
      reflectionApiNames: "反射 API 名单"
    matching:
      preferDirectoryMirror: "优先目录镜像匹配"
      fallbackToNamespaceAndTypeName: "回退命名空间+类型名匹配"
    skeleton:
      enabledByDefault: "默认是否生成骨架（应为 false）"
      requireConfirmation: "是否需用户确认"
      fileCountThreshold: "触发二次确认的文件数阈值"
    realCoverage:
      reportSummaryJson: "报告 Summary.json 路径"
      openCoverRoot: "OpenCover XML 根目录"
      assemblyIncludePrefixes: "纳入采集的程序集前缀"
      assemblyExcludeSuffixes: "排除的程序集后缀"
      branchCoverageAvailable: "分支覆盖率是否可用"
    severity:
      "<issue_kind>": "error/warning/info"
